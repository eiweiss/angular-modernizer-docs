    Interface LibraryExtractionRuleConfig

    Config shape accepted from the .angular-modernizer.json rules section (rules['architecture:library-extraction-opportunity']).

    The analysis rule passes minCohesionScore, minReusabilityScore, targetFolders, categoryOverrides and excludePatterns to the LibraryExtractionScanner; the other fields are not read by the rule.

    interface LibraryExtractionRuleConfig {
        autoGenerateReadme?: boolean;
        categoryOverrides?: CategoryOverrides;
        excludePatterns?: string[];
        incrementalMode?: boolean;
        libraryPrefix?: string;
        libraryType?: "angular-library" | "folder-move";
        maxMigrationComplexity?: "high" | "medium" | "low";
        minCohesionScore?: number;
        minReusabilityScore?: number;
        outputFolder?: string;
        secondaryEntryPointStrategy?:
            | "per-component"
            | "logical-grouping"
            | "folder-based";
        targetFolders?: string[];
    }
