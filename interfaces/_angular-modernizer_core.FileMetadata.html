    Interface FileMetadata

    Metadata about a changed file. Provides context for AI analysis and decision-making.

    Example

    {
      size: { old: 1250, new: 1420 },
      lines: { old: 45, new: 52 },
      encoding: 'utf-8',
      language: 'typescript',
      permissions: { old: '644', new: '644' },
      lastModified: '2025-12-22T10:30:00Z',
      confidence: 0.95
    }
    interface FileMetadata {
        confidence: number;
        encoding: string;
        language: string;
        lastModified?: string;
        lines: { new: number; old: number };
        permissions?: { new: string; old: string };
        size: { new: number; old: number };
    }
