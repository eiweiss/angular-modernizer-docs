    Interface DiffHunk

    Contiguous chunk of changes in a file. Represents a unified diff hunk with rich metadata.

    Example

    {
      oldStart: 10,
      oldLines: 5,
      newStart: 10,
      newLines: 7,
      header: '@@ -10,5 +10,7 @@ export class UserComponent',
      changes: [...],
      summary: {
        additions: 2,
        deletions: 0,
        contextLines: 5
      }
    }
    interface DiffHunk {
        changes: LineChange[];
        containingBlock?: string;
        header: string;
        newLines: number;
        newStart: number;
        oldLines: number;
        oldStart: number;
        summary: { additions: number; contextLines: number; deletions: number };
    }
