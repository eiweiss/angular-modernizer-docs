    Interface FileSystemAdapter

    Core abstraction for file system operations.

    All operations are asynchronous. The implementations (RealFileSystemAdapter, InMemoryFileSystemAdapter) add synchronous variants for ts-morph FileSystemHost compatibility, and the in-memory variant lets tests run without disk access.

    interface FileSystemAdapter {
        createDirectory(path: string): Promise<void>;
        deleteFile(path: string): Promise<void>;
        directoryExists(path: string): Promise<boolean>;
        exists(path: string): Promise<boolean>;
        fileExists(path: string): Promise<boolean>;
        getFileStats(path: string): Promise<{ mtime: number; size: number }>;
        readDirectory(path: string): Promise<string[]>;
        readFile(path: string): Promise<string>;
        writeFile(path: string, content: string): Promise<void>;
    }
