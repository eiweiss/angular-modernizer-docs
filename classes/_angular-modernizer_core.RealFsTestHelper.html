    Class RealFsTestHelper

    Test helper that creates a temporary real directory and a ts-morph project on it, as an alternative to the in-memory file system when a test must reflect real file system behavior (for example schematics tests). RealFsTestHelper.cleanup removes the directory.

    Example

    describe('NgModuleManager (Real FS)', () => {
      let testFs: RealFsTestHelper;

      beforeEach(async () => {
        testFs = await RealFsTestHelper.create();
      });

      afterEach(async () => {
        await testFs.cleanup();
      });

      it('extracts imports', async () => {
        const file = await testFs.createFile('/module.ts', `
          @NgModule({ imports: [CommonModule, FormsModule] })
        `);

        const imports = manager.getNgModuleImports(file);
        expect(imports).toEqual(['CommonModule', 'FormsModule']);
      });
    });
